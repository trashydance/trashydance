import { withAuth } from 'next-auth/middleware';
import { NextResponse } from 'next/server';

export default withAuth(
  function middleware(req) {
    // Qui puoi aggiungere logica extra in futuro
    // es. bloccare utenti con ruolo "banned"
    // const token = req.nextauth.token;
    // if (token?.role === 'banned') return NextResponse.redirect(new URL('/login', req.url));
    return NextResponse.next();
  },
  {
    callbacks: {
      // authorized viene chiamato PRIMA di caricare la pagina
      // se restituisce false → redirect automatico a /login
      authorized: ({ token }) => !!token,
    },
  }
);

FAQ:
"Perché SQLite e non PostgreSQL subito?" → Prisma astrare il DB. Cambiare database in futuro è 1 riga nel schema.prisma + URL nel .env, zero altro codice da toccare.
"Il form login deve fare una fetch manuale all'API?" → No. signIn("credentials", { email, password }) di NextAuth gestisce tutto: manda i dati al backend, riceve il cookie, ricarica la sessione.
"Perché non mettere if (!session) in ogni pagina invece del middleware?" → Il middleware gira server-side prima ancora che la pagina venga renderizzata. È più sicuro (non espone mai il contenuto) e si scrive una volta sola per tutte le rotte.
"Perché bcrypt per le password?" → Il subject lo richiede esplicitamente: nessuna password in chiaro. bcrypt applica salting + hashing irreversibile — anche se il DB venisse compromesso, le password non sarebbero leggibili.
